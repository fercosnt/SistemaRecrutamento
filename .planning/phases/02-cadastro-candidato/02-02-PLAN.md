    ```sql
    -- =============================================================================
    -- Migration: Rate limit for check_candidato_duplicate + policy_version on autorizacoes
    -- Date: 2026-04-21
    -- Requirements: CAD-03, CAD-05 (Phase 2), D-09, D-12, D-16
    -- Threats mitigated: T-02-01 (Info Disclosure — CPF/email enumeration)
    -- =============================================================================
    --
    -- PURPOSE
    -- 1. Add rate_limit_check_duplicate audit table (IP + optional user_id + timestamp)
    --    — consumed ONLY by the SECURITY DEFINER function; REVOKEd from all client roles.
    -- 2. Patch public.check_candidato_duplicate to enforce 30 calls / 60s composite
    --    (ip OR user_id) budget. Excess returns
    --    { cpf_exists: null, email_exists: null, rate_limited: true }.
    -- 3. Add public.autorizacoes.policy_version (D-16) with default 'v1.0-2026-04'.
    --    Defensive-add ip_aceite, data_aceite, user_id (Edge Function already inserts
    --    these best-effort; the baseline migration lacked them per 02-AUDIT-RESULTS.md).
    --
    -- RESPONSE SHAPE (unchanged contract, new field)
    --   { "cpf_exists": boolean|null, "email_exists": boolean|null, "rate_limited": boolean }
    --
    -- CONSUMERS
    --   src/features/cadastro/services/duplicateCheckService.ts (RPC call)
    --   supabase/functions/cadastrar-candidato/index.ts (INSERT into autorizacoes)
    -- =============================================================================

    BEGIN;

    -- -------------------------------------------------------------------------
    -- 1. Rate limit audit table
    -- -------------------------------------------------------------------------

    CREATE TABLE IF NOT EXISTS public.rate_limit_check_duplicate (
      id         bigserial PRIMARY KEY,
      ip         inet NOT NULL,
      user_id    uuid NULL,
      called_at  timestamptz NOT NULL DEFAULT now()
    );

    CREATE INDEX IF NOT EXISTS idx_rate_limit_check_duplicate_ip_recent
      ON public.rate_limit_check_duplicate (ip, called_at DESC);

    CREATE INDEX IF NOT EXISTS idx_rate_limit_check_duplicate_user_recent
      ON public.rate_limit_check_duplicate (user_id, called_at DESC)
      WHERE user_id IS NOT NULL;

    -- Only the SECURITY DEFINER function (running as postgres) reads/writes this table.
    REVOKE ALL ON TABLE public.rate_limit_check_duplicate FROM PUBLIC;
    REVOKE ALL ON TABLE public.rate_limit_check_duplicate FROM anon;
    REVOKE ALL ON TABLE public.rate_limit_check_duplicate FROM authenticated;

    COMMENT ON TABLE public.rate_limit_check_duplicate IS
      'Audit log for check_candidato_duplicate RPC. Rows > 5min old are cleaned '
      'opportunistically inside the function body. Only the SECURITY DEFINER '
      'function may read/write; all client roles are REVOKEd. (D-12, Phase 2)';

    -- -------------------------------------------------------------------------
    -- 2. Patched RPC with embedded rate limit
    -- -------------------------------------------------------------------------

    CREATE OR REPLACE FUNCTION public.check_candidato_duplicate(
      p_cpf   text,
      p_email text
    )
    RETURNS jsonb
    LANGUAGE plpgsql
    SECURITY DEFINER
    SET search_path = ''
    AS $$
    DECLARE
      v_cpf_clean    text;
      v_email_clean  text;
      v_ip           inet;
      v_user_id      uuid;
      v_recent_count integer;
      result         jsonb;
    BEGIN
      v_cpf_clean   := regexp_replace(COALESCE(p_cpf, ''), '\D', '', 'g');
      v_email_clean := lower(trim(COALESCE(p_email, '')));
      v_ip          := inet_client_addr();
      v_user_id     := auth.uid();  -- null for anon; OK

      -- 2a. Rate-limit check: 30 calls / 60s per (ip OR user_id) composite
      SELECT count(*) INTO v_recent_count
      FROM public.rate_limit_check_duplicate
      WHERE (ip = v_ip OR (v_user_id IS NOT NULL AND user_id = v_user_id))
        AND called_at > (now() - interval '60 seconds');

      IF v_recent_count >= 30 THEN
        RETURN jsonb_build_object(
          'cpf_exists',   null,
          'email_exists', null,
          'rate_limited', true
        );
      END IF;

      -- 2b. Log this call
      INSERT INTO public.rate_limit_check_duplicate (ip, user_id, called_at)
      VALUES (v_ip, v_user_id, now());

      -- 2c. Opportunistic cleanup (rows > 5 min old)
      DELETE FROM public.rate_limit_check_duplicate
      WHERE called_at < (now() - interval '5 minutes');

      -- 2d. Normal duplicate check
      SELECT jsonb_build_object(
        'cpf_exists', CASE
          WHEN v_cpf_clean = '' THEN false
          ELSE EXISTS (SELECT 1 FROM public.candidatos
                       WHERE cpf = v_cpf_clean AND deleted_at IS NULL)
        END,
        'email_exists', CASE
          WHEN v_email_clean = '' THEN false
          ELSE EXISTS (SELECT 1 FROM public.candidatos
                       WHERE lower(email) = v_email_clean AND deleted_at IS NULL)
        END,
        'rate_limited', false
      ) INTO result;

      RETURN result;
    END;
    $$;

    COMMENT ON FUNCTION public.check_candidato_duplicate(text, text) IS
      'Returns {cpf_exists, email_exists, rate_limited}. SECURITY DEFINER with '
      'embedded 30 calls / 60s per (IP, user_id) rate limit via '
      'rate_limit_check_duplicate. Opportunistic cleanup deletes rows > 5min old '
      'on each call. (D-09, D-12, Phase 2)';

    -- Grants: keep Phase 1 posture
    REVOKE ALL ON FUNCTION public.check_candidato_duplicate(text, text) FROM PUBLIC;
    GRANT EXECUTE ON FUNCTION public.check_candidato_duplicate(text, text) TO anon, authenticated;

    -- -------------------------------------------------------------------------
    -- 3. policy_version + defensive audit columns on autorizacoes
    -- -------------------------------------------------------------------------

    ALTER TABLE public.autorizacoes
      ADD COLUMN IF NOT EXISTS policy_version text NOT NULL DEFAULT 'v1.0-2026-04';

    ALTER TABLE public.autorizacoes
      ADD COLUMN IF NOT EXISTS ip_aceite inet NULL;

    ALTER TABLE public.autorizacoes
      ADD COLUMN IF NOT EXISTS data_aceite timestamptz NULL DEFAULT now();

    ALTER TABLE public.autorizacoes
      ADD COLUMN IF NOT EXISTS user_id uuid NULL REFERENCES auth.users(id) ON DELETE SET NULL;

    COMMENT ON COLUMN public.autorizacoes.policy_version IS
      'LGPD policy version at time of consent (D-16, Phase 2). Default v1.0-2026-04 '
      'matches supabase/functions/_shared/constants.ts:POLICY_VERSION and '
      'src/features/cadastro/constants.ts:POLICY_VERSION — bump all three together.';

    COMMIT;
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → RPC | Anonymous browser invokes `public.check_candidato_duplicate` — primary enumeration surface |
| postgres → rate_limit_check_duplicate table | Only the SECURITY DEFINER function (running as postgres/owner) may read/write; anon and authenticated are REVOKEd |
| Edge Function → autorizacoes | Service_role INSERTs policy_version; no client path writes this column |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Information Disclosure | `check_candidato_duplicate` RPC | mitigate | Embedded rate limit 30 calls / 60s per (ip ∨ user_id) via rate_limit_check_duplicate table — excess returns `{rate_limited: true}` with nulled booleans. Attacker with 1 IP limited to ~43K probes/day (down from unlimited). |
| T-02-02 | Tampering | `policy_version` column | mitigate | Column is only WRITTEN by the Edge Function (service_role) with the server-side constant `POLICY_VERSION`. Client cannot pass a version — D-16. RLS (existing from Phase 1) prevents client UPDATE. |
| T-02-03 | Information Disclosure | `rate_limit_check_duplicate` table (IPs + user_ids = PII-adjacent) | mitigate | REVOKE ALL on anon + authenticated + PUBLIC. Only the SECURITY DEFINER function's owner (postgres) touches it. Opportunistic cleanup of rows > 5min keeps the retention window minimal. |
| T-02-04 | Denial of Service | `rate_limit_check_duplicate` table unbounded growth | mitigate | Inline `DELETE WHERE called_at < now() - 5 minutes` on every RPC call. Amortized cost is O(1) per call; worst-case bounded by traffic-rate × 5min. |
| T-02-05 | Tampering | RPC function body via `CREATE OR REPLACE` | accept | Requires postgres owner privilege; only reachable via migration applied by operator. Supabase audit log captures the DDL event. |
</threat_model>
