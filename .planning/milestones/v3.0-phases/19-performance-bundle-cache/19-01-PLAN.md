```typescript
// src/router/lazyNamed.ts
import { lazy, type ComponentType } from 'react'
export function lazyNamed<T extends Record<string, ComponentType<any>>>(
  loader: () => Promise<T>,
  name: keyof T,
) {
  return lazy(() => loader().then((m) => ({ default: m[name] })))
}
```
