```typescript
import { z } from 'https://esm.sh/zod@3'

// CPF validation (digit verifier)
function validateCPF(cpf: string): boolean {
  const cleaned = cpf.replace(/\D/g, '')
  if (cleaned.length !== 11) return false
  if (/^(\d)\1+$/.test(cleaned)) return false
  // Digit verification algorithm
  let sum = 0
  for (let i = 0; i < 9; i++) sum += parseInt(cleaned.charAt(i)) * (10 - i)
  let remainder = (sum * 10) % 11
  if (remainder === 10) remainder = 0
  if (remainder !== parseInt(cleaned.charAt(9))) return false
  sum = 0
  for (let i = 0; i < 10; i++) sum += parseInt(cleaned.charAt(i)) * (11 - i)
  remainder = (sum * 10) % 11
  if (remainder === 10) remainder = 0
  return remainder === parseInt(cleaned.charAt(10))
}

export const cadastroCandidatoSchema = z.object({
  // Auth
  email: z.string().email('Email invalido').toLowerCase().trim(),
  password: z.string().min(6, 'Senha deve ter no minimo 6 caracteres'),

  // Dados pessoais
  nome_completo: z.string().min(1, 'Nome e obrigatorio').trim(),
  cpf: z.string().refine(validateCPF, { message: 'CPF invalido' }),
  telefone: z.string().min(1, 'Telefone e obrigatorio'),
  data_nascimento: z.string().min(1, 'Data de nascimento e obrigatoria'),
  genero: z.string().optional(),

  // Endereco
  endereco: z.object({
    cep: z.string().optional(),
    logradouro: z.string().optional(),
    numero: z.string().optional(),
    complemento: z.string().optional(),
    bairro: z.string().optional(),
    cidade: z.string().optional(),
    estado: z.string().optional(),
  }).optional(),

  // Disponibilidade
  disponibilidade: z.object({
    turnos: z.array(z.string()).optional(),
    dias_semana: z.array(z.string()).optional(),
    inicio_imediato: z.boolean().optional(),
  }).optional(),

  // Autorizacoes LGPD
  autorizacoes: z.object({
    aceite_termos: z.boolean().refine(val => val === true, { message: 'Aceite dos termos e obrigatorio' }),
    aceite_lgpd: z.boolean().refine(val => val === true, { message: 'Aceite LGPD e obrigatorio' }),
    aceite_comunicacao: z.boolean().optional(),
  }),
})
```

**Create `supabase/functions/cadastrar-candidato/index.ts` (per D-01, D-01b):**

```typescript
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
import { cadastroCandidatoSchema } from '../_shared/schemas.ts'

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

Deno.serve(async (req) => {
  // Handle CORS preflight
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    const body = await req.json()

    // Validate input
    const parsed = cadastroCandidatoSchema.safeParse(body)
    if (!parsed.success) {
      return new Response(
        JSON.stringify({ ok: false, error: parsed.error.errors[0]?.message || 'Dados invalidos' }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
      )
    }

    const input = parsed.data

    // Create admin client (service_role stays server-side only)
    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!,
      { auth: { autoRefreshToken: false, persistSession: false } }
    )

    // Step 1: Create auth user
    const { data: authData, error: authError } = await supabaseAdmin.auth.admin.createUser({
      email: input.email,
      password: input.password,
      email_confirm: true,
      user_metadata: {
        nome_completo: input.nome_completo,
        cpf: input.cpf,
      },
    })

    if (authError || !authData.user) {
      return new Response(
        JSON.stringify({ ok: false, error: authError?.message || 'Erro ao criar usuario' }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
      )
    }

    const userId = authData.user.id

    try {
      // Step 2: Insert candidato
      const { data: candidatoData, error: candidatoError } = await supabaseAdmin
        .from('candidatos')
        .insert({
          user_id: userId,
          nome_completo: input.nome_completo,
          email: input.email,
          cpf: input.cpf.replace(/\D/g, ''),
          telefone: input.telefone,
          data_nascimento: input.data_nascimento,
          genero: input.genero || null,
          // Endereco fields if the table has them directly, or insert into separate table
          ...(input.endereco ? {
            cep: input.endereco.cep,
            logradouro: input.endereco.logradouro,
            numero: input.endereco.numero,
            complemento: input.endereco.complemento,
            bairro: input.endereco.bairro,
            cidade: input.endereco.cidade,
            estado: input.endereco.estado,
          } : {}),
        })
        .select('id')
        .single()

      if (candidatoError || !candidatoData) {
        throw new Error(`Insert candidato failed: ${candidatoError?.message}`)
      }

      // Step 3: Insert disponibilidade (if provided and table exists)
      if (input.disponibilidade) {
        await supabaseAdmin
          .from('disponibilidade_candidato')
          .insert({
            candidato_id: candidatoData.id,
            ...input.disponibilidade,
          })
          .then(({ error }) => {
            if (error) console.error('Disponibilidade insert warning:', error.message)
          })
      }

      // Step 4: Insert autorizacoes LGPD
      await supabaseAdmin
        .from('autorizacoes')
        .insert({
          candidato_id: candidatoData.id,
          user_id: userId,
          aceite_termos: input.autorizacoes.aceite_termos,
          aceite_lgpd: input.autorizacoes.aceite_lgpd,
          aceite_comunicacao: input.autorizacoes.aceite_comunicacao || false,
          ip_aceite: req.headers.get('x-forwarded-for') || 'unknown',
          data_aceite: new Date().toISOString(),
        })
        .then(({ error }) => {
          if (error) console.error('Autorizacoes insert warning:', error.message)
        })

      return new Response(
        JSON.stringify({ ok: true, data: { candidatoId: candidatoData.id, userId } }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
      )
    } catch (innerError) {
      // Rollback: delete the auth user if subsequent inserts fail
      await supabaseAdmin.auth.admin.deleteUser(userId).catch(() => {})
      return new Response(
        JSON.stringify({ ok: false, error: (innerError as Error).message }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 500 }
      )
    }
  } catch (error) {
    return new Response(
      JSON.stringify({ ok: false, error: (error as Error).message }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 400 }
    )
  }
})
```
