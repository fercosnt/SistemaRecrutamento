  /**
   * Auto-login após cadastro bem-sucedido. Single retry with 500ms backoff (D-02).
   * Returns true iff at least one attempt succeeds. NEVER logs the password.
   */
  export async function tryAutoLogin(email: string, password: string): Promise<boolean> {
    const first = await supabase.auth.signInWithPassword({ email, password })
    if (!first.error) return true
    await new Promise((r) => setTimeout(r, 500))
    const second = await supabase.auth.signInWithPassword({ email, password })
    return !second.error
  }
