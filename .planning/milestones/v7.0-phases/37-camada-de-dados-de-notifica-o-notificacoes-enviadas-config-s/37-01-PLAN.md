-- (A) COLUNAS — ordem, tipo, nulidade, default
select table_name, ordinal_position, column_name, data_type, udt_name,
       is_nullable, column_default
  from information_schema.columns
 where table_schema = 'public'
   and table_name in ('notificacoes_enviadas','config_sla_etapa')
 order by table_name, ordinal_position;

-- (B) CONSTRAINTS — nome + definição canônica do Postgres
select rel.relname as tabela, con.conname, con.contype,
       pg_get_constraintdef(con.oid) as definicao
  from pg_constraint con
  join pg_class rel on rel.oid = con.conrelid
  join pg_namespace n on n.oid = rel.relnamespace
 where n.nspname = 'public'
   and rel.relname in ('notificacoes_enviadas','config_sla_etapa')
 order by rel.relname, con.conname;

-- (C) ÍNDICES — definição canônica
select tablename, indexname, indexdef
  from pg_indexes
 where schemaname = 'public'
   and tablename in ('notificacoes_enviadas','config_sla_etapa')
 order by tablename, indexname;

-- (D) POLICIES — roles, cmd, qual, with_check  ⚠ o literal de role sai daqui
select tablename, policyname, permissive, roles, cmd, qual, with_check
  from pg_policies
 where schemaname = 'public'
   and tablename in ('notificacoes_enviadas','config_sla_etapa')
 order by tablename, policyname;

-- (E) RLS habilitada + contagem de triggers NÃO-internos
select c.relname, c.relrowsecurity, c.relforcerowsecurity,
       (select count(*) from pg_trigger t
         where t.tgrelid = c.oid and not t.tgisinternal) as triggers_nao_internos
  from pg_class c
  join pg_namespace n on n.oid = c.relnamespace
 where n.nspname = 'public'
   and c.relname in ('notificacoes_enviadas','config_sla_etapa');

-- (F) ENUMS — labels em ordem
select t.typname, e.enumsortorder, e.enumlabel
  from pg_type t
  join pg_enum e on e.enumtypid = t.oid
  join pg_namespace n on n.oid = t.typnamespace
 where n.nspname = 'public'
   and t.typname in ('status_notificacao','etapa_processo')
 order by t.typname, e.enumsortorder;

-- (G) COMMENTs de tabela e de coluna
select c.relname, obj_description(c.oid,'pg_class') as comentario_tabela
  from pg_class c join pg_namespace n on n.oid = c.relnamespace
 where n.nspname = 'public'
   and c.relname in ('notificacoes_enviadas','config_sla_etapa');
select c.relname, a.attname, col_description(c.oid, a.attnum) as comentario_coluna
  from pg_class c
  join pg_namespace n on n.oid = c.relnamespace
  join pg_attribute a on a.attrelid = c.oid and a.attnum > 0 and not a.attisdropped
 where n.nspname = 'public'
   and c.relname in ('notificacoes_enviadas','config_sla_etapa')
 order by c.relname, a.attnum;

-- (H) GRANTs de tabela por role
select table_name, grantee, privilege_type
  from information_schema.role_table_grants
 where table_schema = 'public'
   and table_name in ('notificacoes_enviadas','config_sla_etapa')
 order by table_name, grantee, privilege_type;

-- (I) COLISÃO DE NOME da função de trigger que a 37-03 vai criar
select n.nspname, p.proname,
       pg_get_function_identity_arguments(p.oid) as args,
       p.prosecdef, p.proconfig
  from pg_proc p join pg_namespace n on n.oid = p.pronamespace
 where n.nspname = 'public'
   and p.proname in ('tocar_atualizado_em','update_updated_at_column');

-- (J) DADOS — premissa "0 linhas" da 37-03 + seed completo
select count(*) as linhas_notificacoes from public.notificacoes_enviadas;
select count(*) as linhas_sla from public.config_sla_etapa;
select etapa, prazo_valor, prazo_unidade, rotulo_candidato, atualizado_em
  from public.config_sla_etapa order by etapa;

-- (K) LEDGER — prova de que os reconstruídos NÃO devem ser re-aplicados
select version
  from supabase_migrations.schema_migrations
 where version in ('20260721000001','20260721000002','20260722000001','20260722000002')
 order by version;
